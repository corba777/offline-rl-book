% Preprint look in the spirit of arXiv templates: Times-like text and math,
% a small-caps running head, compact chapter and section headings.
\usepackage{fontspec}
\usepackage{unicode-math}
\setmainfont{STIXTwoText}[
  Extension = .otf,
  UprightFont = *-Regular,
  BoldFont = *-Bold,
  ItalicFont = *-Italic,
  BoldItalicFont = *-BoldItalic]
\setmathfont{STIXTwoMath-Regular.otf}
\setmonofont{DejaVuSansMono}[
  Extension = .ttf,
  UprightFont = *,
  BoldFont = *-Bold,
  ItalicFont = *-Oblique,
  BoldItalicFont = *-BoldOblique,
  Scale = 0.8]

% Symbols that STIX Two Text lacks: take them from STIX Two Math or DejaVu Sans
% in running text; code keeps DejaVu Sans Mono, which has them.
\usepackage{newunicodechar}
\newfontfamily\textsymbols{STIXTwoMath-Regular.otf}
\newfontfamily\circledsymbols{DejaVuSans.ttf}
\makeatletter
\newcommand{\symchar}[2]{\edef\sym@fam{\f@family}\edef\sym@tt{\ttdefault}%
  \ifx\sym@fam\sym@tt\char#1\relax\else{#2\char#1\relax}\fi}
\makeatother
\newunicodechar{→}{\symchar{"2192}{\textsymbols}}
\newunicodechar{≈}{\symchar{"2248}{\textsymbols}}
\newunicodechar{≠}{\symchar{"2260}{\textsymbols}}
\newunicodechar{①}{\symchar{"2460}{\circledsymbols}}
\newunicodechar{②}{\symchar{"2461}{\circledsymbols}}
\newunicodechar{③}{\symchar{"2462}{\circledsymbols}}

\usepackage{microtype}
\usepackage{titlesec}
\titleformat{\chapter}[hang]{\normalfont\LARGE\bfseries}{\thechapter}{0.8em}{}
\titlespacing*{\chapter}{0pt}{0pt}{1.4em}
\titleformat{\section}{\normalfont\large\bfseries}{\thesection}{0.8em}{}
\titleformat{\subsection}{\normalfont\normalsize\bfseries}{\thesubsection}{0.8em}{}
\titleformat{\subsubsection}{\normalfont\normalsize\bfseries\itshape}{\thesubsubsection}{0.8em}{}

\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[C]{\small\scshape Offline RL: From Theory to Industrial Practice\quad--\quad A Preprint}
\fancyfoot[C]{\small\thepage}
\renewcommand{\headrulewidth}{0pt}
\fancypagestyle{plain}{\fancyhf{}\fancyfoot[C]{\small\thepage}\renewcommand{\headrulewidth}{0pt}}

% Long code lines wrap instead of running off the page.
\usepackage{fvextra}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{breaklines,breakanywhere,commandchars=\\\{\}}
\fvset{fontsize=\small}

% Figures sit where the text puts them.
\usepackage{float}
\floatplacement{figure}{H}

% Wide tables shrink a little.
\AtBeginEnvironment{longtable}{\small}
\setlength{\LTpre}{0.6em}
\setlength{\LTpost}{0.6em}

\setlength{\parskip}{0.45em}
\setlength{\emergencystretch}{3em}
\setlength{\parindent}{0pt}
\renewcommand{\arraystretch}{1.15}
